cmake_minimum_required(VERSION 3.16)
project(xdna_package_consumer LANGUAGES C CXX)

if (NOT PACKAGE_TEST_MODE MATCHES "^(STATIC|DL|OFF)$" OR NOT IS_DIRECTORY "${PACKAGE_PREFIX}")
    message(FATAL_ERROR "Set PACKAGE_TEST_MODE to STATIC/DL/OFF and PACKAGE_PREFIX to the installation")
endif()
if (DEFINED GGML_XDNA_XRT_ROOT)
    set(_consumer_xrt_root_defined TRUE)
    set(_consumer_xrt_root "${GGML_XDNA_XRT_ROOT}")
endif()
find_package(ggml CONFIG REQUIRED PATHS "${PACKAGE_PREFIX}/lib/cmake/ggml" NO_DEFAULT_PATH)
if (_consumer_xrt_root_defined AND NOT "${GGML_XDNA_XRT_ROOT}" STREQUAL "${_consumer_xrt_root}")
    message(FATAL_ERROR "ggml overwrote the consumer XRT root")
endif()
get_filename_component(_prefix "${PACKAGE_PREFIX}" REALPATH)
function(require_installed_path path)
    get_filename_component(_actual "${path}" REALPATH)
    string(FIND "${_actual}" "${_prefix}/" _position)
    if (NOT _position EQUAL 0)
        message(FATAL_ERROR "Imported ggml path escaped the installation: ${_actual}")
    endif()
    message(STATUS "Installed package path: ${_actual}")
endfunction()
require_installed_path("${ggml_CONFIG}")
require_installed_path("${GGML_INCLUDE_DIR}")
set(_targets ggml::ggml ggml::ggml-base)
if (NOT GGML_BACKEND_DL)
    foreach(_backend IN LISTS GGML_AVAILABLE_BACKENDS)
        list(APPEND _targets ggml::${_backend})
    endforeach()
endif()
foreach(_target IN LISTS _targets)
    get_target_property(_location ${_target} IMPORTED_LOCATION)
    require_installed_path("${_location}")
endforeach()
if (PACKAGE_TEST_MODE STREQUAL "STATIC")
    if (NOT GGML_XDNA OR GGML_SHARED_LIB OR GGML_BACKEND_DL OR NOT TARGET XRT::xrt_coreutil)
        message(FATAL_ERROR "Expected static ggml/XDNA and its transitive XRT target")
    endif()
    message(STATUS "Consumer XRT root: ${GGML_XDNA_XRT_ROOT}; XRT config: ${XRT_DIR}")
elseif (PACKAGE_TEST_MODE STREQUAL "DL")
    if (NOT GGML_XDNA OR NOT GGML_SHARED_LIB OR NOT GGML_BACKEND_DL OR TARGET XRT::xrt_coreutil)
        message(FATAL_ERROR "Expected dynamic backends without a consumer XRT SDK dependency")
    endif()
else()
    if (GGML_XDNA OR GGML_SHARED_LIB OR GGML_BACKEND_DL OR TARGET XRT::xrt_coreutil)
        message(FATAL_ERROR "Expected an XDNA-disabled static package without XRT")
    endif()
endif()
add_executable(package-consumer main.cpp)
target_compile_features(package-consumer PRIVATE cxx_std_17)
target_compile_definitions(package-consumer PRIVATE PACKAGE_${PACKAGE_TEST_MODE})
target_link_libraries(package-consumer PRIVATE ggml::ggml ${CMAKE_DL_LIBS})
